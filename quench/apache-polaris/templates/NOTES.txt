  CLIENT_ID=$(kubectl get secret {{ include "apache-polaris.authSecret" . }} -n {{ .Release.Namespace }} -o jsonpath='{.data.client-id}' | base64 -d)
  CLIENT_SECRET=$(kubectl get secret {{ include "apache-polaris.authSecret" . }} -n {{ .Release.Namespace }} -o jsonpath='{.data.client-secret}' | base64 -d)

  kubectl port-forward svc/{{ include "quench-common.fullname" . }} 8181:{{ .Values.service.port }}

    -d client_id=$CLIENT_ID -d client_secret=$CLIENT_SECRET -d scope=PRINCIPAL_ROLE:ALL
